// helpers.ts
export const heroEnter = (frame, fps, delay=4) => {
  const p = spring({frame, fps, config: MOTION.heroSpring});
  const breath = Math.sin((frame-(delay+14))*0.045)*0.012;
  return { scale: p*(1+breath), translateY:(1-p)*40, opacity:p };
};
// Scene01 TYPE_TIMINGS — hard cut, no crossfade
抽走复用：最先产品化：做成可配置字幕轨组件，再套进任意 Scene。
